cmake_minimum_required(VERSION 3.16)
project(eye_reference LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(test_reference tests/test_reference.cpp)
target_compile_features(test_reference PRIVATE cxx_std_17)
target_include_directories(test_reference PRIVATE include)
target_link_libraries(test_reference PRIVATE Threads::Threads)
enable_testing()
add_test(NAME eye_reference_behavior COMMAND test_reference)
